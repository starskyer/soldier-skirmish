cmake_minimum_required(VERSION 3.21)
project(SoldierSkirmish VERSION 0.2.0 LANGUAGES CXX)
option(SKIRMISH_BUILD_GAME "Build the SDL windowed game" ON)
option(SKIRMISH_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(skirmish-options INTERFACE)
target_compile_features(skirmish-options INTERFACE cxx_std_17)
if(MSVC)
    target_compile_options(skirmish-options INTERFACE /W4 /utf-8)
else()
    target_compile_options(skirmish-options INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(SKIRMISH_SANITIZERS)
    if(WIN32 OR MSVC)
        message(FATAL_ERROR "Sanitizer configuration is supported by the Linux CI job")
    endif()
    target_compile_options(skirmish-options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(skirmish-options INTERFACE -fsanitize=address,undefined)
endif()
function(skirmish_target name)
    set_target_properties(${name} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    target_link_libraries(${name} PRIVATE skirmish-options)
endfunction()
add_library(skirmish-core STATIC
    src/chessboard.cpp src/base.cpp src/soldier.cpp src/soldier_move.cpp
    src/attackBehavior.cpp src/turnResolver.cpp src/model.cpp src/vision.cpp src/frameClock.cpp)
target_include_directories(skirmish-core PUBLIC src)
skirmish_target(skirmish-core)
add_executable(soldier-skirmish-sim src/sim_main.cpp src/appOptions.cpp)
target_link_libraries(soldier-skirmish-sim PRIVATE skirmish-core)
skirmish_target(soldier-skirmish-sim)
add_custom_target(copy-assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:soldier-skirmish-sim>/assets")
add_dependencies(soldier-skirmish-sim copy-assets)
if(SKIRMISH_BUILD_GAME)
    find_package(SDL2 2.0.18 CONFIG REQUIRED)
    find_package(SDL2_image 2.0 CONFIG REQUIRED)
    find_package(SDL2_ttf 2.0 CONFIG REQUIRED)
    add_executable(soldier-skirmish src/main.cpp src/runtimeAssets.cpp src/controller.cpp src/view.cpp src/appOptions.cpp)
    target_compile_definitions(soldier-skirmish PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(soldier-skirmish PRIVATE skirmish-core SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
    skirmish_target(soldier-skirmish)
    add_dependencies(soldier-skirmish copy-assets)
    if(WIN32)
        add_custom_command(TARGET soldier-skirmish POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_RUNTIME_DLLS:soldier-skirmish> $<TARGET_FILE_DIR:soldier-skirmish>
            COMMAND_EXPAND_LISTS)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(skirmish-core-tests tests/core_tests.cpp)
    target_link_libraries(skirmish-core-tests PRIVATE skirmish-core)
    skirmish_target(skirmish-core-tests)
    add_test(NAME core COMMAND skirmish-core-tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
    add_test(NAME full-match COMMAND soldier-skirmish-sim --ticks 18000 --seed 1 --require-finished)
    add_test(NAME full-match-seed-2 COMMAND soldier-skirmish-sim --ticks 18000 --seed 2 --require-finished)
    add_test(NAME full-match-seed-42 COMMAND soldier-skirmish-sim --ticks 18000 --seed 42 --require-finished)
    set_tests_properties(core full-match full-match-seed-2 full-match-seed-42 PROPERTIES LABELS core TIMEOUT 60)
    if(SKIRMISH_BUILD_GAME)
        add_test(NAME assets COMMAND soldier-skirmish --check-assets)
        add_test(NAME startup COMMAND soldier-skirmish --smoke-test)
        set_tests_properties(startup PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 15)
    endif()
endif()
